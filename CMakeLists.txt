cmake_minimum_required(VERSION 3.20)
project(conrad_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cjson STATIC third_party/cjson/cJSON.c)
target_include_directories(cjson PUBLIC third_party/cjson)
add_library(robot STATIC firmware/components/robot/robot.cpp firmware/components/robot/service.cpp firmware/components/robot/dm_codec.cpp)
target_include_directories(robot PUBLIC firmware/components/robot/include)
target_link_libraries(robot PUBLIC cjson)
if(MSVC)
  target_compile_options(robot PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(robot PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(robot_sim host/main.cpp)
target_link_libraries(robot_sim PRIVATE robot)
add_executable(robot_tests tests/core_tests.cpp)
target_link_libraries(robot_tests PRIVATE robot)
enable_testing()
add_test(NAME robot_core COMMAND robot_tests)
